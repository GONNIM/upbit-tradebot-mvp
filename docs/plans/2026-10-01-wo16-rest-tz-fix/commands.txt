# ===== wo16_parse.py =====
"""WO-16 사전 측정: VERIFY 실행별 건너뛴 봉 파싱.
입력: wo16_verify_raw.log (journal 추출, 09-24 ~ 10-01)
출력: wo16_skips.csv (run_time, range_start, range_end, skip_ts, pos_from_oldest) + 분포 요약
"""
import collections
import csv
import re
from datetime import datetime

S = "/private/tmp/claude-501/-Users-gonnim-Project-MVP-Source-upbit-tradebot-mvp/9339ae21-5ead-4004-a49c-c5714e34e251/scratchpad"
re_start = re.compile(r"^(\S+ \S+) .*\[VERIFY\] 과거 봉 검증 시작 \| count=(\d+) \| range=(\S+ \S+) KST ~ (\S+ \S+) KST")
re_skip = re.compile(r"^(\S+ \S+) .*Upbit에 없는 timestamp \| ts=(\S+ \S+) KST")
fmt = "%Y-%m-%d %H:%M:%S"

runs = []
cur = None
for line in open(f"{S}/wo16_verify_raw.log", encoding="utf-8"):
    m = re_start.match(line)
    if m:
        cur = {"t": m.group(1), "count": int(m.group(2)), "a": m.group(3), "b": m.group(4), "skips": []}
        runs.append(cur)
        continue
    m = re_skip.match(line)
    if m and cur is not None:
        cur["skips"].append(m.group(2))

rows = []
for r in runs:
    a = datetime.strptime(r["a"], fmt)
    for s in r["skips"]:
        pos = int((datetime.strptime(s, fmt) - a).total_seconds() // 300)
        rows.append([r["t"], r["a"], r["b"], s, pos])

with open(f"{S}/wo16_skips.csv", "w", newline="") as f:
    w = csv.writer(f)
    w.writerow(["run_time", "range_start", "range_end", "skip_ts", "slot_from_range_start"])
    w.writerows(rows)

n_runs = len(runs)
n_skip = sum(len(r["skips"]) for r in runs)
dist = collections.Counter(len(r["skips"]) for r in runs)
print("runs", n_runs, "skips", n_skip, "unique bars", len({x[3] for x in rows}))
print("skips per run (top):", sorted(dist.items())[:12], "... max", max(dist))
# 건너뛴 봉이 범위 앞쪽(오래된 쪽)에 몰려 있는지: 연속 앞부분인지
contig = 0
for r in runs:
    if not r["skips"]:
        continue
    srt = sorted(r["skips"])
    if srt[0] == r["a"]:
        contig += 1
print("runs whose first skip == range_start:", contig, "/", sum(1 for r in runs if r["skips"]))
big = [r for r in runs if len(r["skips"]) > 5]
print("runs with >5 skips:", len(big))
for r in big[:3]:
    print(r["t"], r["a"], r["b"], len(r["skips"]), r["skips"][:3], r["skips"][-2:])
by_day = collections.Counter(r["t"][:10] for r in big)
print("big runs by day:", sorted(by_day.items()))

# ===== wo16_match.py =====
"""WO-16 사전 측정: VERIFY 가 건너뛴 봉(09-24 ~ 10-01) vs Upbit 확정 OHLC vs 엔진 기록 값.

Upbit: 공개 캔들 API, to=UTC 'Z' (배타), 200봉씩. 키 불필요.
엔진 기록 값: audit_buy_eval / audit_sell_eval (읽기 전용 추출 wo16_audit.csv)
  - price          : 실시간 평가 때 쓴 종가
  - backfill_close : BACKFILL 재평가 때 쓴 종가(REST 값)
출력: verify_gap_match.csv (봉 단위), 요약 출력
"""
import collections
import csv
import json
import time
import urllib.request
from datetime import datetime, timedelta, timezone

S = "/private/tmp/claude-501/-Users-gonnim-Project-MVP-Source-upbit-tradebot-mvp/9339ae21-5ead-4004-a49c-c5714e34e251/scratchpad"
OUT = "/Users/gonnim/Project-MVP/Source/upbit-tradebot-mvp/docs/plans/2026-10-01-wo16-rest-tz-fix/verify_gap_match.csv"
KST = timezone(timedelta(hours=9))
TOL = 1.0  # VERIFY tolerance ±1원

# ---- 1. 건너뛴 봉 (실행 단위) ----
runs = collections.defaultdict(list)          # skip_ts -> [(run_time, range_start, range_end)]
for r in csv.DictReader(open(f"{S}/wo16_skips.csv")):
    runs[r["skip_ts"]].append((r["run_time"], r["range_start"], r["range_end"]))

# ---- 2. Upbit 확정 캔들 (09-21 ~ 10-01 13:00 KST) ----
upbit = {}
to = datetime(2026, 10, 1, 13, 0, tzinfo=KST).astimezone(timezone.utc)
stop = datetime(2026, 9, 21, 0, 0, tzinfo=KST).astimezone(timezone.utc)
while to > stop:
    url = ("https://api.upbit.com/v1/candles/minutes/5?market=KRW-JTO&count=200&to="
           + to.strftime("%Y-%m-%dT%H:%M:%SZ"))
    data = json.load(urllib.request.urlopen(urllib.request.Request(url, headers={"accept": "application/json"}), timeout=10))
    if not data:
        break
    for c in data:
        k = c["candle_date_time_kst"].replace("T", " ")
        upbit[k] = (c["opening_price"], c["high_price"], c["low_price"], c["trade_price"], c["candle_acc_trade_volume"])
    to = min(datetime.strptime(c["candle_date_time_utc"], "%Y-%m-%dT%H:%M:%S") for c in data).replace(tzinfo=timezone.utc)
    time.sleep(0.15)

# ---- 3. 엔진 기록 값 ----
audit = collections.defaultdict(list)
for r in csv.DictReader(open(f"{S}/wo16_audit.csv")):
    k = r["bar_time"][:19].replace("T", " ")
    audit[k].append(r)


def f(x):
    try:
        return float(x)
    except (TypeError, ValueError):
        return None


rows = []
cls_count = collections.Counter()
diff_dist = collections.Counter()
for ts in sorted(runs):
    rs = runs[ts]
    unsorted = any(a > b for _, a, b in rs)                 # 범위 시작 > 끝 = 로컬 시계열 정렬 깨짐
    edge = all(ts == a for _, a, _b in rs if a <= _b)       # 정렬된 실행에서는 늘 범위 첫 봉
    u = upbit.get(ts)
    recs = audit.get(ts, [])
    rt = [f(r["price"]) for r in recs if f(r["price"]) is not None and not r["backfill_type"]]
    rt_any = [f(r["price"]) for r in recs if f(r["price"]) is not None]
    bf = [f(r["backfill_close"]) for r in recs if f(r["backfill_close"]) is not None]
    local_rt = rt[0] if rt else (rt_any[0] if rt_any else None)
    local_bf = bf[-1] if bf else None
    uc = u[3] if u else None

    if u is None:
        cls = "upbit_no_candle"
    elif local_rt is None and local_bf is None:
        cls = "no_engine_record"
    else:
        d_rt = None if local_rt is None else local_rt - uc
        d_bf = None if local_bf is None else local_bf - uc
        if (d_bf is not None and abs(d_bf) <= TOL) or (d_bf is None and abs(d_rt) <= TOL):
            cls = "match"
        elif d_bf is None and abs(d_rt) > TOL:
            cls = "rt_mismatch_only"     # 실시간 평가 종가만 다름(그 뒤 REST 덮어쓰기)
        else:
            cls = "backfill_mismatch"
    cls_count[cls] += 1
    if u is not None and local_rt is not None:
        diff_dist[round(local_rt - uc)] += 1

    rows.append([
        ts, len(rs), rs[0][0], rs[-1][0], "Y" if unsorted else "N", "Y" if edge and not unsorted else "N",
        *(u if u else ("", "", "", "", "")),
        "" if local_rt is None else local_rt,
        "" if local_bf is None else local_bf,
        "" if (u is None or local_rt is None) else round(local_rt - uc, 4),
        "" if (u is None or local_bf is None) else round(local_bf - uc, 4),
        ";".join(sorted({r["tbl"] + ":" + (r["backfill_type"] or "rt") for r in recs})),
        cls,
    ])

with open(OUT, "w", newline="") as fo:
    w = csv.writer(fo)
    w.writerow(["skip_ts_kst", "n_runs_skipped", "first_run", "last_run", "in_unsorted_run", "oldest_edge_only",
                "upbit_open", "upbit_high", "upbit_low", "upbit_close", "upbit_volume",
                "engine_rt_close", "engine_backfill_close", "diff_rt", "diff_backfill", "audit_rows", "class"])
    w.writerows(rows)

print("upbit bars fetched", len(upbit), min(upbit), max(upbit))
print("unique skipped bars", len(rows))
print("class", dict(cls_count))
print("rt diff dist", sorted(diff_dist.items()))
print("unsorted bars", sum(1 for r in rows if r[4] == "Y"), "edge-only bars", sum(1 for r in rows if r[5] == "Y"))
mm = [r for r in rows if r[-1] not in ("match",)]
for r in mm[:12]:
    print(r)

# ===== wo16_to_probe.py =====
"""WO-16: Upbit 캔들 API `to` 경계 실측 (공개 API, 키 불필요, 읽기 전용).
기준 봉: 2026-10-01 12:00 KST (= 03:00 UTC), 5분봉.
"""
import json
import time
import urllib.parse
import urllib.request

BASE = "https://api.upbit.com/v1/candles/minutes/5?market=KRW-JTO&count={n}"


def q(to=None, n=1):
    url = BASE.format(n=n) + ("" if to is None else "&to=" + urllib.parse.quote(to))
    d = json.load(urllib.request.urlopen(urllib.request.Request(url, headers={"accept": "application/json"}), timeout=10))
    time.sleep(0.15)
    return [c["candle_date_time_kst"] for c in d]


cases = [
    ("P1 to=03:00:00Z (봉 시작 시각 정확히)", "2026-10-01T03:00:00Z"),
    ("P2 to=03:00:01Z (봉 시작 +1초)", "2026-10-01T03:00:01Z"),
    ("P3 to=03:05:00Z (봉 시작 + 봉 간격)", "2026-10-01T03:05:00Z"),
    ("P4 to='2026-10-01 03:05:00' (시간대 없음, UTC 값)", "2026-10-01 03:05:00"),
    ("P5 to='2026-10-01 12:05:00' (시간대 없음, KST 값 — 현 코드 방식)", "2026-10-01 12:05:00"),
    ("P6 to='2026-10-01T12:05:00+09:00' (KST 오프셋 명시)", "2026-10-01T12:05:00+09:00"),
]
for name, to in cases:
    print(f"{name:55s} -> {q(to)}")
print(f"{'P7 to 없음 (count=2)':55s} -> {q(None, 2)}")
