# ===== wo17_ast_hash.py =====
"""WO-17 (C): 함수 실행 코드(AST, docstring 제외) 해시 계산. 주석은 AST 에 없으므로 주석 변경은 해시 불변."""
import ast
import hashlib
import sys

TARGETS = [
    ("core/rest_reconcile.py", "fetch_confirmed_candle"),
    ("core/data_feed.py", "stream_candles"),
]


def exec_hash(path, name):
    tree = ast.parse(open(path, encoding="utf-8").read())
    for node in ast.walk(tree):
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)) and node.name == name:
            for n in ast.walk(node):
                body = getattr(n, "body", None)
                if isinstance(body, list) and body and isinstance(body[0], ast.Expr) \
                        and isinstance(getattr(body[0], "value", None), ast.Constant) \
                        and isinstance(body[0].value.value, str):
                    body.pop(0)
            return hashlib.sha256(ast.dump(node, include_attributes=False).encode()).hexdigest()
    return None


if __name__ == "__main__":
    for p, n in TARGETS:
        print(p, n, exec_hash(p, n))

#!/bin/bash
# WO-18·17 커밋별 단독 revert 실측 (임시 worktree, main 무변경)
set -u
REPO=/Users/gonnim/Project-MVP/Source/upbit-tradebot-mvp
WT=/private/tmp/claude-501/-Users-gonnim-Project-MVP-Source-upbit-tradebot-mvp/9339ae21-5ead-4004-a49c-c5714e34e251/scratchpad/wo18_wt
COMMITS=(8bf7944 17742cb 83f48ba)

git -C "$REPO" worktree remove --force "$WT" 2>/dev/null
git -C "$REPO" worktree add --detach "$WT" HEAD >/dev/null 2>&1 || { echo "worktree 생성 실패"; exit 1; }
# 게이트가 .env 를 읽지 않도록 worktree 에는 .env 를 두지 않는다 (git 미추적 파일이라 복사되지 않음)
ls -a "$WT" | grep -c '^\.env$' | sed 's/^/worktree .env 개수: /'
# git 미추적 .streamlit(secrets) 은 링크, 업비트 키는 더미 환경변수 (실키 미사용)
ln -s "$REPO/.streamlit" "$WT/.streamlit"
export UPBIT_ACCESS=dummy UPBIT_SECRET=dummy TELEGRAM_BOT_TOKEN= TELEGRAM_CHAT_ID=

for c in "${COMMITS[@]}"; do
  echo "=================== revert $c ($(git -C "$REPO" log --format=%s -1 "$c" | cut -c1-40))"
  git -C "$WT" reset -q --hard HEAD
  git -C "$WT" clean -qfd
  git -C "$WT" revert --no-commit "$c" >/dev/null 2>&1
  conflicts=$(git -C "$WT" diff --name-only --diff-filter=U)
  echo "충돌 파일: ${conflicts:-없음}"
  other=$(echo "$conflicts" | grep -v '^pages/dashboard.py$' | grep -v '^$')
  if [ -n "$other" ]; then
    echo "❌ 버전 줄 외 충돌: $other"
    git -C "$WT" diff "$other" | head -40
    continue
  fi
  if echo "$conflicts" | grep -q '^pages/dashboard.py$'; then
    git -C "$WT" checkout --ours pages/dashboard.py
    git -C "$WT" add pages/dashboard.py
    echo "버전 줄 충돌 → 현재 버전 유지: $(grep -o 'v1\.2026\.[0-9.]*' "$WT/pages/dashboard.py" | head -1)"
  fi
  echo "변경 파일:"; git -C "$WT" diff --cached --stat | tail -5
  (cd "$WT" && python3 -m py_compile core/rest_reconcile.py core/data_feed.py core/indicator_state.py core/strategy_engine.py core/position_state.py services/db.py engine/live_loop.py pages/dashboard.py && echo "py_compile OK")
  (cd "$WT" && bash scripts/regression_gate.sh 2>&1 | grep -E "회귀 테스트 .*통과|실패|FAILED|❌" | tail -3)
done

git -C "$WT" reset -q --hard HEAD
git -C "$REPO" worktree remove --force "$WT"
echo "worktree 정리 완료"; git -C "$REPO" worktree list
