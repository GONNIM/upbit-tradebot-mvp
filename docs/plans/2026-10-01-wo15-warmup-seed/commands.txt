#!/bin/bash
# WO-15 읽기 전용 집계 (journal 2026-08-30 이후)
SINCE="2026-08-30"
G() { journalctl -u tradebot --since "$SINCE" --no-pager -g "$1" 2>/dev/null | grep -vF "fragment with id"; }

echo "== (Q2) 다중 호출 요청 수 → 수신 수 분포"
journalctl -u tradebot --since "$SINCE" --no-pager -g "다중 호출 시작|다중 호출 완료" 2>/dev/null \
  | sed -E 's/^.*\]: //' \
  | awk '/다중 호출 시작/{match($0,/total_count=[0-9]+/); req=substr($0,RSTART+12,RLENGTH-12); next}
         /다중 호출 완료/{match($0,/total=[0-9]+/); got=substr($0,RSTART+6,RLENGTH-6); match($0,/batches=[0-9]+/); b=substr($0,RSTART+8,RLENGTH-8); print req" -> "got" (batches="b")"}' \
  | sort | uniq -c | sort -rn | head -15

echo "== (Q3) VERIFY 'Upbit에 없는 timestamp' 경고 건수: $(G 'Upbit에 없는 timestamp' | wc -l)"
echo "== (Q3) VERIFY 검증 시작 건수: $(G 'VERIFY\] 과거 봉 검증 시작 \| count=.*현재 봉' | wc -l)"
echo "== (Q3) VERIFY 실패(현재 봉 스킵) 건수: $(G '과거 봉 검증 실패' | wc -l)"

G '재조회 성공' | sed -E 's/^.*\]: //' | cut -c1-150 | tail -5
echo "== (Q1) 부분 재계산 시작 중 bars>=200: $(G '부분 재계산 시작' | grep -cE 'bars=([2-9][0-9]{2}|[0-9]{4,})')"

# ===== wo15_fetch.py =====
"""WO-15: Upbit 공개 캔들 API(키 불필요, 읽기 전용)로 KRW-JTO 5분봉 수집.
to 는 UTC 'Z' 명시. 결과: wo15_candles.csv (ts_kst, open, high, low, close, volume)
"""
import csv
import json
import time
import urllib.request
from datetime import datetime, timedelta, timezone

KST = timezone(timedelta(hours=9))
OUT = "/private/tmp/claude-501/-Users-gonnim-Project-MVP-Source-upbit-tradebot-mvp/9339ae21-5ead-4004-a49c-c5714e34e251/scratchpad/wo15_candles.csv"

# 끝: 2026-10-01 13:00 KST (exclusive), 1200봉(약 4일 + 무거래 봉 누락분)
to = datetime(2026, 10, 1, 13, 0, tzinfo=KST).astimezone(timezone.utc)
rows = {}
for _ in range(6):
    url = (
        "https://api.upbit.com/v1/candles/minutes/5?market=KRW-JTO&count=200&to="
        + to.strftime("%Y-%m-%dT%H:%M:%SZ")
    )
    req = urllib.request.Request(url, headers={"accept": "application/json"})
    data = json.load(urllib.request.urlopen(req, timeout=10))
    if not data:
        break
    for c in data:
        rows[c["candle_date_time_kst"]] = (
            c["opening_price"], c["high_price"], c["low_price"],
            c["trade_price"], c["candle_acc_trade_volume"],
        )
    oldest_utc = min(datetime.strptime(c["candle_date_time_utc"], "%Y-%m-%dT%H:%M:%S") for c in data)
    to = oldest_utc.replace(tzinfo=timezone.utc)  # Upbit to 는 exclusive → 가장 오래된 봉 시각 그대로
    time.sleep(0.2)

with open(OUT, "w", newline="") as f:
    w = csv.writer(f)
    w.writerow(["ts_kst", "open", "high", "low", "close", "volume"])
    for k in sorted(rows):
        w.writerow([k, *rows[k]])
print(len(rows), min(rows), max(rows))

# ===== wo15_drift.py =====
"""WO-15 (1): 워밍업 형성 중 봉 시드 오차 수치화.

엔진 재현 (core/indicator_state.py):
  seed_from_closes: EMA = 마지막 N개 종가의 SMA (N=60, 200)
  update_incremental: ema = a*close + (1-a)*ema, a = 2/(N+1)
  recompute_from_changed_ts: tail < 200 이면 seed 실패 → 상태 변화 없음 (로그로 확인)
  BACKFILL 11:40 재평가: 지표 백업·복원 → 상태 변화 없음

세 계열 비교:
  engine   : 2026-10-01 11:42:08 시드 (워밍업 200봉, 마지막 11:40 봉 = 형성 중 종가 729)
  fixseed  : 같은 200봉이되 11:40 봉을 확정 종가(733)로 시드
  ref      : 1200봉 장기 이력으로 계산한 EMA (참고: SMA 시드 자체의 구조 오차 비교용)
"""
import csv

SCR = "/private/tmp/claude-501/-Users-gonnim-Project-MVP-Source-upbit-tradebot-mvp/9339ae21-5ead-4004-a49c-c5714e34e251/scratchpad"
OUT = "/Users/gonnim/Project-MVP/Source/upbit-tradebot-mvp/docs/plans/2026-10-01-wo15-warmup-seed/seed_drift.csv"

rows = list(csv.DictReader(open(f"{SCR}/wo15_candles.csv")))
ts = [r["ts_kst"] for r in rows]
close = {r["ts_kst"]: float(r["close"]) for r in rows}

WARM_START, WARM_END = "2026-09-30T18:55:00", "2026-10-01T11:40:00"
LAST_RT = "2026-10-01T12:20:00"   # 관측 마지막 실시간 봉 (Bar#207)
FORMING_CLOSE = 729.0             # 11:42:08 워밍업이 받은 11:40 봉 형성 중 종가 (로그)

warm = [t for t in ts if WARM_START <= t <= WARM_END]
assert len(warm) == 200, len(warm)
rt = [t for t in ts if WARM_END < t <= LAST_RT]

def sma(xs, n):
    return sum(xs[-n:]) / n

def alpha(n):
    return 2 / (n + 1)

warm_confirmed = [close[t] for t in warm]
warm_engine = warm_confirmed[:-1] + [FORMING_CLOSE]

# 장기 기준: 1200봉 처음부터 SMA 시드 후 증분 (11:40 까지)
def long_ema(n):
    first = ts[:n]
    e = sma([close[t] for t in first], n)
    for t in ts[n:]:
        if t > WARM_END:
            break
        e = alpha(n) * close[t] + (1 - alpha(n)) * e
    return e

state = {
    "engine": {60: sma(warm_engine, 60), 200: sma(warm_engine, 200)},
    "fixseed": {60: sma(warm_confirmed, 60), 200: sma(warm_confirmed, 200)},
    "ref": {60: long_ema(60), 200: long_ema(200)},
}

header = [
    "k", "bar_ts_kst", "close",
    "ema60_engine", "ema60_fixseed", "ema60_ref", "d60_engine_minus_fixseed", "d60_engine_minus_ref",
    "ema200_engine", "ema200_fixseed", "ema200_ref", "d200_engine_minus_fixseed", "d200_engine_minus_ref",
    "gap_engine(60-200)", "gap_fixseed(60-200)", "gap_ref(60-200)",
]
out = []

def snap(k, t, c):
    e, f, r = state["engine"], state["fixseed"], state["ref"]
    out.append([
        k, t, c,
        round(e[60], 4), round(f[60], 4), round(r[60], 4), round(e[60] - f[60], 4), round(e[60] - r[60], 4),
        round(e[200], 4), round(f[200], 4), round(r[200], 4), round(e[200] - f[200], 4), round(e[200] - r[200], 4),
        round(e[60] - e[200], 4), round(f[60] - f[200], 4), round(r[60] - r[200], 4),
    ])

snap(0, "seed@11:40 (engine close=729 / confirmed 733)", close[WARM_END])
for k, t in enumerate(rt, start=1):
    c = close[t]
    for s in state.values():
        for n in (60, 200):
            s[n] = alpha(n) * c + (1 - alpha(n)) * s[n]
    snap(k, t, c)

# 형성 중 봉 오차의 이론 감쇠 (잔차 = 초기 오차 × (1-a)^k): 1% 와 0.1% 로 줄어드는 봉 수
import math
for n in (60, 200):
    a = alpha(n)
    out.append([])
    out.append([f"# EMA{n}: a={a:.5f}, 초기 시드 오차 = (729-733)/{n} = {(-4)/n:.4f}원, "
                f"1%로 감쇠 {math.ceil(math.log(0.01)/math.log(1-a))}봉, "
                f"0.1%로 감쇠 {math.ceil(math.log(0.001)/math.log(1-a))}봉"])

with open(OUT, "w", newline="") as f:
    w = csv.writer(f)
    w.writerow(header)
    w.writerows(out)

for r in out[:9]:
    print(r)
for r in out[-5:]:
    print(r)

"""WO-15 (1) 보강: 기동별 워밍업 마지막 봉(형성 중 여부)과 시드 오차.

"""

# (기동 시 [REST] 최신 확정 봉 로그 시각, 마지막 봉 ts, 봉 간격 분, 워밍업 수신 종가) — journal 원문
BOOTS = [

    ("2026-09-18 17:26:34", "2026-09-18 17:20:00", 5, 623),
    ("2026-09-25 12:41:06", "2026-09-25 12:40:00", 5, 667),
    ("2026-09-30 16:33:06", "2026-09-30 16:30:00", 5, 755),
    ("2026-09-30 18:02:09", "2026-09-30 18:00:00", 5, 759),
    ("2026-09-30 19:44:29", "2026-09-30 19:40:00", 5, 755),
    ("2026-10-01 09:56:09", "2026-10-01 09:50:00", 5, 727),
    ("2026-10-01 11:42:08", "2026-10-01 11:40:00", 5, 729),
]


def confirmed_close(bar_kst: datetime, minutes: int) -> float:
    to = (bar_kst + timedelta(minutes=minutes)).astimezone(timezone.utc)
    url = (f"https://api.upbit.com/v1/candles/minutes/{minutes}?market=KRW-JTO&count=1&to="
           + to.strftime("%Y-%m-%dT%H:%M:%SZ"))
    d = json.load(urllib.request.urlopen(urllib.request.Request(url, headers={"accept": "application/json"}), timeout=10))
    assert d and d[0]["candle_date_time_kst"] == bar_kst.strftime("%Y-%m-%dT%H:%M:%S"), d
    time.sleep(0.15)
    return float(d[0]["trade_price"])


rows = []
for boot_s, last_s, m, got in BOOTS:
    boot = datetime.strptime(boot_s, "%Y-%m-%d %H:%M:%S").replace(tzinfo=KST)
    last = datetime.strptime(last_s, "%Y-%m-%d %H:%M:%S").replace(tzinfo=KST)
    forming = boot < last + timedelta(minutes=m)
    conf = confirmed_close(last, m)
    diff = got - conf
    elapsed = int((boot - last).total_seconds())
    rows.append([
        boot_s, last_s, f"minute{m}", "형성 중" if forming else "확정",
        elapsed if forming else "", got, conf, diff,
        round(diff / 60, 4), round(diff / 200, 4), round(diff / 60 - diff / 200, 4),
        round(100 * diff / conf, 3),
    ])

with open(OUT, "a", newline="") as f:

    w.writerow(["# 표 2. 기동별 워밍업 마지막 봉과 시드 오차 (2026-08-30 이후 journal 의 워밍업 13회, 전부 '마지막 봉 유지')"])
    w.writerow(["boot_kst", "last_bar_kst", "interval", "last_bar_state", "elapsed_in_bar_sec",
                "warmup_close", "confirmed_close", "diff",
                "ema60_seed_err", "ema200_seed_err", "gap_seed_err(60-200)", "diff_pct"])
    w.writerows(rows)

for r in rows:
    print(r)
