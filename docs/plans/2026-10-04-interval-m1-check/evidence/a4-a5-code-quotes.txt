    def interval_sec(self) -> int:
        """
        interval 문자열을 초(sec) 단위로 변환.
        예: "minute1" -> 60, "minute3" -> 180, "minute5" -> 300

        interval_map = {
            "minute1": 60,
            "minute3": 180,
            "minute5": 300,
            "minute10": 600,
            "minute15": 900,
            "minute30": 1800,
            "minute60": 3600,
            "day": 86400,
        }
        return interval_map.get(self.interval, 60)  # 기본값 60초
