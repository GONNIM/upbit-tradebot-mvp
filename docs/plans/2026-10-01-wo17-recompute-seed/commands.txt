"""G4: Upbit 공개 캔들 API(키 불필요) 수집. to = UTC 'Z', 봉 시작 시각 기준 배타(WO-16 실측).

열: ts_kst, open, high, low, close, volume
"""
import csv
import json
import time
import urllib.request
from datetime import datetime, timedelta, timezone

KST = timezone(timedelta(hours=9))


def fetch(unit, start_kst, end_kst, out):
    rows, to = {}, end_kst.astimezone(timezone.utc)
    stop = start_kst.astimezone(timezone.utc)
    calls = 0
    while to > stop:
        url = (f"https://api.upbit.com/v1/candles/minutes/{unit}?market=KRW-JTO&count=200&to="
               + to.strftime("%Y-%m-%dT%H:%M:%SZ"))
        for attempt in range(5):
            try:
                d = json.load(urllib.request.urlopen(urllib.request.Request(url, headers={"accept": "application/json"}), timeout=15))
                break
            except Exception as e:  # 429 등
                time.sleep(1.0 + attempt)
        else:
            raise RuntimeError(url)
        calls += 1
        if not d:
            break
        for c in d:
            rows[c["candle_date_time_kst"]] = (c["opening_price"], c["high_price"], c["low_price"],
                                               c["trade_price"], c["candle_acc_trade_volume"])
        to = min(datetime.strptime(c["candle_date_time_utc"], "%Y-%m-%dT%H:%M:%S") for c in d).replace(tzinfo=timezone.utc)
        time.sleep(0.12)
    with open(out, "w", newline="") as f:
        w = csv.writer(f)
        w.writerow(["ts_kst", "open", "high", "low", "close", "volume"])
        for k in sorted(rows):
            if k >= start_kst.strftime("%Y-%m-%dT%H:%M:%S"):
                w.writerow([k, *rows[k]])
    print(out, len(rows), "calls", calls, min(rows), max(rows))

def run(seed_bars, seed_kind, p, stream):
    """seed_kind: 'sma_last' (A) | 'sma_first_then_inc' (B, C). stream: 이후 (ts, close). 반환 {ts: ema} + 시드 값."""
    closes = [c for _, c in seed_bars]
    a = 2 / (p + 1)
    if seed_kind == "sma_last":
        e = sum(closes[-p:]) / p
    else:
        e = sum(closes[:p]) / p
        for c in closes[p:]:
            e = a * c + (1 - a) * e
    seed = e
    out = {}
    for ts, c in stream:
        e = a * c + (1 - a) * e
        out[ts] = e
    return seed, out


def crosses(seed_f, seed_s, f, s, eval_ts):
    """eval_ts 순서대로 교차. prev = 직전 봉(첫 봉은 시드 상태)."""
    ev = []
    pf, ps = seed_f, seed_s
    all_ts = sorted(f)
    for ts in all_ts:
        cf, cs = f[ts], s[ts]
        if ts in eval_ts:
            if pf <= ps and cf > cs:
                ev.append((ts, "GC"))
            elif pf >= ps and cf < cs:
                ev.append((ts, "DC"))
        pf, ps = cf, cs
    return ev


def pair(evx, evy, idx, maxgap=36):
    """같은 종류 교차를 가까운 것끼리 짝. 반환 [(type, tx, ty, shift_bars, status)]."""
    used, res = set(), []
    for tx, k in evx:
        best = None
        for j, (ty, k2) in enumerate(evy):
            if j in used or k2 != k:
                continue
            d = idx[ty] - idx[tx]
            if abs(d) <= maxgap and (best is None or abs(d) < abs(best[1])):
                best = (j, d)
        if best:
            used.add(best[0])
            res.append((k, tx, evy[best[0]][0], best[1], "matched"))
        else:
            res.append((k, tx, None, None, "only_x"))
    for j, (ty, k2) in enumerate(evy):
        if j not in used:
            res.append((k2, None, ty, None, "only_y"))
    return res

# ===== wo17s_verify_seed.py =====
"""WO-17 (S) 배포 검증용: 기동 시각 기준 장기 기준(확정 봉 1,200봉 이상) EMA60/EMA200 재계산.
사용: python3 wo17s_verify_seed.py "2026-10-0X HH:MM:SS"(기동 [WARMUP] 시각, KST) <log_fast> <log_slow> [bars=1200]
- Upbit 공개 캔들 API(키 불필요), to = UTC 'Z', 봉 시작 시각 기준 배타.
- 확정 봉(봉 시작 + 5분 <= 기동 시각)만. 첫 200봉 SMA → 나머지 증분(구현과 같은 식).
- 같은 시각 800봉(구현 재현)도 함께 계산해 로그 값과 비교.
"""
import json
import sys
import time
import urllib.request
from datetime import datetime, timedelta, timezone

KST = timezone(timedelta(hours=9))
boot = datetime.strptime(sys.argv[1], "%Y-%m-%d %H:%M:%S").replace(tzinfo=KST)
log_fast, log_slow = float(sys.argv[2]), float(sys.argv[3])
N = int(sys.argv[4]) if len(sys.argv) > 4 else 1200

bars, to = {}, boot.astimezone(timezone.utc)
while len(bars) < N + 50:
    url = ("https://api.upbit.com/v1/candles/minutes/5?market=KRW-JTO&count=200&to=" + to.strftime("%Y-%m-%dT%H:%M:%SZ"))
    d = json.load(urllib.request.urlopen(urllib.request.Request(url, headers={"accept": "application/json"}), timeout=15))
    if not d:
        break
    for c in d:
        t = datetime.strptime(c["candle_date_time_utc"], "%Y-%m-%dT%H:%M:%S").replace(tzinfo=timezone.utc)
        bars[t] = float(c["trade_price"])
    to = min(bars)
    time.sleep(0.15)
conf = [(t, c) for t, c in sorted(bars.items()) if t + timedelta(minutes=5) <= boot]


def seed(closes, base=200):
    out = {}
    for p in (60, 200):
        a = 2 / (p + 1)
        e = sum(closes[base - p:base]) / p          # seed_from_closes(first 200): 마지막 p개 SMA
        for c in closes[base:]:
            e = a * c + (1 - a) * e
        out[p] = e
    return out


long_ = seed([c for _, c in conf[-N:]])
s800 = seed([c for _, c in conf[-800:]])
print(f"마지막 확정 봉 KST: {conf[-1][0].astimezone(KST):%Y-%m-%d %H:%M}  bars_long={len(conf[-N:])}")
print(f"로그        fast={log_fast:.4f} slow={log_slow:.4f}")
print(f"800 재현    fast={s800[60]:.4f} slow={s800[200]:.4f}  (로그와 차이 {s800[60]-log_fast:+.4f} / {s800[200]-log_slow:+.4f})")
print(f"{N} 기준  fast={long_[60]:.4f} slow={long_[200]:.4f}  (로그와 차이 {long_[60]-log_fast:+.4f} / {long_[200]-log_slow:+.4f})")
