# 조회 전용: GET /v1/orders/closed (주문 생성·취소 없음)
import os, sys, uuid, hashlib, jwt, requests
from urllib.parse import urlencode, unquote
sys.path.insert(0, "/root/upbit-tradebot-mvp")
os.chdir("/root/upbit-tradebot-mvp")
for line in open(".env", encoding="utf-8"):
    line=line.strip()
    if line and not line.startswith("#") and "=" in line:
        k,v=line.split("=",1); os.environ.setdefault(k.strip(), v.strip().strip('"').strip("'"))
ak, sk = os.getenv("UPBIT_ACCESS"), os.getenv("UPBIT_SECRET")
if not ak:
    import tomllib
    s = tomllib.load(open(".streamlit/secrets.toml", "rb")); ak, sk = s["UPBIT_ACCESS"], s["UPBIT_SECRET"]
def get(path, params):
    q = unquote(urlencode(params, doseq=True)).encode()
    payload = {"access_key": ak, "nonce": str(uuid.uuid4()), "query_hash": hashlib.sha512(q).hexdigest(), "query_hash_alg": "SHA512"}
    tok = jwt.encode(payload, sk)
    r = requests.get("https://api.upbit.com" + path, params=params, headers={"Authorization": f"Bearer {tok}"}, timeout=10)
    return r.status_code, r.json()
for st in (["cancel"], ["done"]):
    code, data = get("/v1/orders/closed", {"market": "KRW-JTO", "states[]": st, "start_time": "2026-09-29T18:00:00+09:00", "end_time": "2026-09-30T08:00:00+09:00", "limit": 100, "order_by": "asc"})
    print("== state", st, "http", code)
    if isinstance(data, list):
        for o in data:
            print(o.get("created_at"), o.get("side"), o.get("ord_type"), o.get("price"), o.get("volume"), o.get("executed_volume"), o.get("state"), o.get("uuid"))
    else:
        print(data)

# Upbit 공개 REST minute5 KRW-JTO 수집 (조회 전용, 인증 불필요)
import requests, time, json, sys
out = []; to = "2026-09-30T08:00:00+09:00"; start = "2026-09-27T00:00:00"
while True:
    r = requests.get("https://api.upbit.com/v1/candles/minutes/5", params={"market": "KRW-JTO", "to": to, "count": 200}, timeout=10)

    if not d: break
    out += d

## 부록: recalc.py
```python
# core/indicator_state.py seed_from_closes 절차 재현: 앞쪽 slow개 SMA 시드 → alpha=2/(N+1) 증분
import json, csv, sys
from datetime import datetime, timedelta
S = sys.argv[1]
c = json.load(open(f"{S}/candles.json"))
raw = [(x["candle_date_time_kst"][:16], float(x["trade_price"])) for x in c]
def ffill(rows):
    d = dict(rows); t = datetime.fromisoformat(rows[0][0]); end = datetime.fromisoformat(rows[-1][0]); out=[]; last=None
    while t <= end:
        k = t.strftime("%Y-%m-%dT%H:%M")
        if k in d: last = d[k]; out.append((k, last, 0))
        else: out.append((k, last, 1))
        t += timedelta(minutes=5)
    return out
def run(rows, fast, slow, start):
    rows = [r for r in rows if r[0] >= start]
    closes = [r[1] for r in rows]
    af, as_ = 2/(fast+1), 2/(slow+1)
    seed = closes[:slow]
    ef = sum(seed[-fast:])/fast; es = sum(seed)/slow
    res = {}
    for i in range(slow, len(rows)):
        p = closes[i]; ef += af*(p-ef); es += as_*(p-es)
        res[rows[i][0]] = (p, ef, es, rows[i][2] if len(rows[i])>2 else 0)
    return res
bot = {r["bar_time"][:16]: r for r in csv.DictReader(open(f"{S}/sell_raw.csv"))}
A_rows = [(k, p, 0) for k, p in raw]
B_rows = ffill(raw)
variants = {
 "A_skip_27": run(A_rows, 60, 200, "2026-09-27T00:00"),
 "B_ffill_27": run(B_rows, 60, 200, "2026-09-27T00:00"),
 "B_ffill_28_18": run(B_rows, 60, 200, "2026-09-28T18:00"),
 "A_skip_28_18": run(A_rows, 60, 200, "2026-09-28T18:00"),
}
for name, v in variants.items():
    df = [abs(v[k][1]-float(b["ema_fast"])) for k, b in bot.items() if k in v]
    ds = [abs(v[k][2]-float(b["ema_slow"])) for k, b in bot.items() if k in v]
    print(f"{name}: matched={len(df)} max|dFast|={max(df):.4f} mean={sum(df)/len(df):.4f} max|dSlow|={max(ds):.4f} mean={sum(ds)/len(ds):.4f}")
json.dump({n: v for n, v in variants.items()}, open(f"{S}/variants.json", "w"))
```

import json, csv, sys
S = sys.argv[1]; exec(open(f"{S}/recalc.py").read().split("bot = {")[0].replace("S = sys.argv[1]", ""))
bot = {r["bar_time"][:16]: r for r in csv.DictReader(open(f"{S}/sell_raw.csv"))}
A_rows = [(k, p, 0) for k, p in raw]
out = []
for fast, slow in ((60, 200), (20, 60)):
    v = run(A_rows, fast, slow, "2026-09-27T00:00")
    keys = sorted(v); prev = None
    for k in keys:
        p, ef, es, _ = v[k]
        if prev and k >= "2026-09-29T18:00" and k <= "2026-09-30T08:00":
            pf, ps = v[prev][1], v[prev][2]
            ev = "DC" if (pf >= ps and ef < es) else ("GC" if (pf < ps and ef >= es) else None)
            if ev:
                b = bot.get(k, {})
                row = dict(pair=f"{fast}/{slow}", event=ev, bar_time_kst=k, close=p, recalc_ema_fast=round(ef,4), recalc_ema_slow=round(es,4),
                    bot_ema_fast=b.get("ema_fast",""), bot_ema_slow=b.get("ema_slow",""),
                    diff_fast=round(ef-float(b["ema_fast"]),4) if b else "", diff_slow=round(es-float(b["ema_slow"]),4) if b else "",
                    bot_cross_status=b.get("cross_status",""), bot_ema_dc_detected=b.get("ema_dc_detected",""), bot_reason=b.get("reason",""))
                out.append(row); print(row)
        prev = k
w = csv.DictWriter(open(f"{S}/recalc_dc_events.csv", "w", newline=""), fieldnames=list(out[0].keys())); w.writeheader(); w.writerows(out)
